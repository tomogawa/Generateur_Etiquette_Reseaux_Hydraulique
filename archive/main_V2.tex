\newcommand{\DetecterPictos}{%
\ifdefempty{\PictoUn}{\EtqPictoUnActiffalse}{\EtqPictoUnActiftrue}%
\ifdefempty{\PictoDeux}{\EtqPictoDeuxActiffalse}{\EtqPictoDeuxActiftrue}%
\ifdefempty{\PictoTrois}{\EtqPictoTroisActiffalse}{\EtqPictoTroisActiftrue}%
}

% ======================================================================
% Calcul des proportions
% ======================================================================
\newcommand{\CalculEtiquette}{%
\DetecterPictos%
\setlength{\EtqHauteur}{\HauteurEtiquette}%

\ifdim\EtqHauteur=50mm
\setlength{\EtqGap}{4mm}%
\else
\ifdim\EtqHauteur=26mm
\setlength{\EtqGap}{8mm}%
\else
\ifdim\EtqHauteur=100mm
\setlength{\EtqGap}{9mm}%
\else
\setlength{\EtqGap}{0.08\EtqHauteur}%
\fi
\fi
\fi

\setlength{\EtqPolice}{0.40\EtqHauteur}%
\setlength{\EtqInterligne}{1.15\EtqPolice}%
\setlength{\EtqPointe}{\RatioPointe\EtqHauteur}%
\setlength{\EtqMarge}{0.30\EtqHauteur}%
\setlength{\EtqDecalage}{0.08\EtqHauteur}%
\setlength{\EtqBord}{0.045\EtqHauteur}%
\setlength{\EtqPictoTaille}{\RatioPicto\EtqHauteur}%
\setlength{\EtqGapPicto}{0.15\EtqHauteur}%

% Largeur cumulée des pictos actifs (taille + espace après chacun)
\setlength{\EtqLargeurPictos}{0mm}%
\ifEtqPictoUnActif
\addtolength{\EtqLargeurPictos}{\EtqPictoTaille+\EtqGapPicto}%
\fi
\ifEtqPictoDeuxActif
\addtolength{\EtqLargeurPictos}{\EtqPictoTaille+\EtqGapPicto}%
\fi
\ifEtqPictoTroisActif
\addtolength{\EtqLargeurPictos}{\EtqPictoTaille+\EtqGapPicto}%
\fi

\sbox0{\fontsize{\EtqPolice}{\EtqInterligne}\selectfont\bfseries\TexteEtiquette}%
\setlength{\EtqLargeurTexte}{\wd0}%

\setlength{\EtqLargeurTotale}{2\EtqDecalage + 2\EtqPointe + \EtqLargeurPictos + \EtqLargeurTexte + 2\EtqMarge + 2\EtqGap}%
}

% ======================================================================
% Dessin de l'étiquette
% ======================================================================
\newcommand{\DessinerEtiquette}{%
\begin{tikzpicture}[x=1mm,y=1mm]
\pgfmathsetmacro{\Hn}{\EtqHauteur/1mm}
\pgfmathsetmacro{\Wn}{\EtqLargeurTotale/1mm}
\pgfmathsetmacro{\Pn}{\EtqPointe/1mm}
\pgfmathsetmacro{\Bn}{\EtqBord/1mm}
\pgfmathsetmacro{\Dn}{\EtqDecalage/1mm}
\pgfmathsetmacro{\Gap}{\EtqGap/1mm}
\pgfmathsetmacro{\Marge}{\EtqMarge/1mm}
\pgfmathsetmacro{\PictoT}{\EtqPictoTaille/1mm}
\pgfmathsetmacro{\GapP}{\EtqGapPicto/1mm}
\useasboundingbox (0,0) rectangle (\Wn,\Hn);

\fill[white] (0,0) rectangle (\Wn,\Hn);

\draw[black,line width=\Bn mm]
(\Dn,{\Hn/2}) -- ({\Dn+\Pn},0) -- ({\Dn+\Pn},\Hn) -- cycle;

\draw[black,line width=\Bn mm]
({\Wn-\Dn},{\Hn/2}) -- ({\Wn-\Dn-\Pn},0) -- ({\Wn-\Dn-\Pn},\Hn) -- cycle;

\draw[black,line width=\Bn mm]
({\Dn+\Pn+\Gap},0) rectangle ({\Wn-\Dn-\Pn-\Gap},\Hn);

% ----- Positionnement du contenu (pictos puis texte) -----
\pgfmathsetmacro{\xContenu}{\Dn+\Pn+\Gap+\Marge}

\ifEtqPictoUnActif
\node[anchor=west,inner sep=0pt] at (\xContenu,{\Hn/2}) {\includegraphics[height=\PictoT mm]{\PictoUn.png}};
\pgfmathsetmacro{\xContenu}{\xContenu+\PictoT+\GapP}
\fi

\ifEtqPictoDeuxActif
\node[anchor=west,inner sep=0pt] at (\xContenu,{\Hn/2}) {\includegraphics[height=\PictoT mm]{\PictoDeux.png}};
\pgfmathsetmacro{\xContenu}{\xContenu+\PictoT+\GapP}
\fi

\ifEtqPictoTroisActif
\node[anchor=west,inner sep=0pt] at (\xContenu,{\Hn/2}) {\includegraphics[height=\PictoT mm]{\PictoTrois.png}};
\pgfmathsetmacro{\xContenu}{\xContenu+\PictoT+\GapP}
\fi

\node[black,anchor=west,font=\fontsize{\EtqPolice}{\EtqInterligne}\selectfont\bfseries]
at (\xContenu,{\Hn/2}) {\TexteEtiquette};
\end{tikzpicture}%
}

\CalculEtiquette%
\setlength{\PageW}{\EtqLargeurTotale+2\EtqMargePage}%

\pdfpagewidth=\PageW\relax
\pdfpageheight=\PageH\relax
\shipout\vbox to \PageH{%
\vfil
\hbox to \PageW{\hfil\DessinerEtiquette\hfil}%
\vfil
}%
